import matplotlib; matplotlib.use('Agg')
import matplotlib.pyplot as plt
import io, base64

plt.style.use('seaborn-v0_8-whitegrid')
fig, ax = plt.subplots(figsize=(10, 6))

regions = df['region'].tolist()
rev_2024 = df['rev_2024'].tolist()
rev_2023 = df['rev_2023'].tolist()

x = range(len(regions))
width = 0.35

colors_2024 = ['#E24B4A' if a else '#4A72B0' for a in df['is_anomaly']]
colors_2023 = ['#99B4D8' if a else '#6A8FC0' for a in df['is_anomaly']]

ax.bar([i - width/2 for i in x], rev_2024, width, label='2024', color=colors_2024)
ax.bar([i + width/2 for i in x], rev_2023, width, label='2023', color=colors_2023)

ax.set_xticks(x)
ax.set_xticklabels(regions)
ax.set_xlabel('Region')
ax.set_ylabel('Revenue (VND)')
ax.set_title('Q4 Revenue Comparison by Region')
ax.legend()
fig.tight_layout()

buf = io.BytesIO()
plt.savefig(buf, format='png', dpi=150, bbox_inches='tight')
buf.seek(0)
b64 = base64.b64encode(buf.read()).decode('utf-8')
plt.close()

result = {"chart_b64": b64, "chart_type": "grouped_bar", "title": "Q4 Revenue Comparison", "x_col": "region", "y_col": "rev_2024"}
